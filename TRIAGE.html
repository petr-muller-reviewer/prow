related-codePopRandom — the randomization primitive

  • where: pkg/layeredsets/string.go:99-111
  • excerpt:
    func (s String) PopRandom() string {
    for _, layer := range s {
        if layer.Len() > 0 {
            list := sets.List(layer)
            sort.Strings(list)
            sel := list[rand.Intn(len(list))]
            s.Delete(sel)
            return sel
        }
    }
    return ""
}
